cmake_minimum_required(VERSION 3.16)
project(modguard_linux VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(exhaust_model STATIC src/exhaust_model.cpp src/config.cpp)
set_target_properties(exhaust_model PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
target_include_directories(exhaust_model PUBLIC src)
target_compile_options(exhaust_model PRIVATE -Wall -Wextra -Wpedantic)

add_library(config_runtime STATIC src/config_runtime.cpp)
set_target_properties(config_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
target_include_directories(config_runtime PUBLIC src)
target_link_libraries(config_runtime PUBLIC exhaust_model)
target_compile_options(config_runtime PRIVATE -Wall -Wextra -Wpedantic)

add_library(config_watcher STATIC src/config_watcher.cpp)
set_target_properties(config_watcher PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
target_include_directories(config_watcher PUBLIC src include)
target_link_libraries(config_watcher PUBLIC config_runtime)
find_package(Threads REQUIRED)
target_link_libraries(config_watcher PRIVATE Threads::Threads)
target_compile_options(config_watcher PRIVATE -Wall -Wextra -Wpedantic)

add_library(particle_hook STATIC src/particle_hook.cpp src/particle_signature.cpp src/particle_admission.cpp src/particle_blend.cpp)
set_target_properties(particle_hook PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
target_include_directories(particle_hook PUBLIC src)
target_compile_options(particle_hook PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(particle_hook PRIVATE dl)
target_link_libraries(particle_hook PUBLIC game_detector)

add_library(game_detector STATIC src/game_detector.cpp src/game_profiles.cpp)
set_target_properties(game_detector PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
target_include_directories(game_detector PUBLIC src)
target_compile_options(game_detector PRIVATE -Wall -Wextra -Wpedantic)

add_library(modguard_linux SHARED src/plugin.cpp)
set_target_properties(modguard_linux PROPERTIES PREFIX "" OUTPUT_NAME "modguard_linux")
set_target_properties(modguard_linux PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
target_include_directories(modguard_linux PRIVATE include src)
target_link_libraries(modguard_linux PRIVATE exhaust_model config_runtime config_watcher particle_hook)
target_link_libraries(modguard_linux PRIVATE Threads::Threads)
target_compile_options(modguard_linux PRIVATE -Wall -Wextra -Wpedantic)

add_executable(modguard_check tools/modguard_check.cpp)
target_link_libraries(modguard_check PRIVATE game_detector)
target_compile_options(modguard_check PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)
